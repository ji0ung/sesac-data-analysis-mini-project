# 2026-08-28 18:00 이후 USER 기준 SQL 정제 코드

## 1. 목적

기존 PowerShell 정제 코드인 `.artifact_work/process_valid_data.ps1`의 처리 조건을 SQLite SQL로 재현한다.

- SQL 방언: SQLite
- 기준 시각: `2026-08-28 18:00:00 KST`
- 기준 모집단: `user.signup_at >= 기준 시각`인 사용자
- 기준 시각 포함 여부: 포함 (`>=`)
- 기준 키: `user_id`
- 원본 테이블: `user`, `event`, `search`, `search_result`, `search_filter`, `booking`
- 결과 테이블: `user_filtered`, `event_filtered`, `search_filtered`, `search_result_filtered`, `search_filter_filtered`, `booking_filtered`

> 아래 SQL은 결과 테이블을 생성하므로 원본 SQLite 파일을 복사한 뒤 복사본에서 실행하는 것을 권장한다. 원본 업무 테이블은 수정하지 않지만, 같은 이름의 기존 `*_filtered` 결과 테이블은 삭제 후 다시 만든다.

## 2. PowerShell과 SQL의 대응 관계

| 기존 PowerShell 처리 | SQLite SQL 처리 |
|---|---|
| `Parse-Kst()` | `datetime(replace(trim(시각컬럼), ' KST', ''))` |
| `$validUserIds` HashSet | TEMP 테이블 `_valid_user_ids` |
| `$sessionUsers` HashSet | TEMP 테이블 `_valid_session_owner` |
| `$validSearchIds` HashSet | TEMP 테이블 `_valid_search_ids` |
| `Where-Object` 필터 | `WHERE` 및 `JOIN` |
| `Export-Csv` | `CREATE TABLE ... AS SELECT ...` |

SQLite에는 별도 시간대 객체를 만들지 않고, 원문 끝의 ` KST`를 제거한 뒤 동일한 KST 시각끼리 비교한다. 입력값의 형식은 `yyyy-MM-dd HH:mm:ss KST` 또는 `yyyy-MM-dd HH:mm:ss`여야 한다.

## 3. 전체 실행 SQL

```sql
-- ============================================================
-- 2026-08-28 18:00:00 KST 이후 가입 사용자 기준 데이터 정제
-- 대상 DBMS: SQLite
-- ============================================================

PRAGMA foreign_keys = ON;

BEGIN IMMEDIATE;

-- ------------------------------------------------------------
-- 0. 이전 실행 결과 정리
--    원본 테이블은 건드리지 않고 결과 테이블만 다시 만든다.
-- ------------------------------------------------------------

DROP TABLE IF EXISTS user_filtered;
DROP TABLE IF EXISTS event_filtered;
DROP TABLE IF EXISTS search_filtered;
DROP TABLE IF EXISTS search_result_filtered;
DROP TABLE IF EXISTS search_filter_filtered;
DROP TABLE IF EXISTS booking_filtered;

DROP TABLE IF EXISTS temp._valid_user_ids;
DROP TABLE IF EXISTS temp._valid_session_owner;
DROP TABLE IF EXISTS temp._valid_search_ids;

-- ------------------------------------------------------------
-- 1. USER 코호트 선정
--    signup_at이 2026-08-28 18:00:00 KST 이상인 사용자만 선정한다.
--    NULL 또는 파싱할 수 없는 signup_at은 조건을 통과하지 못한다.
-- ------------------------------------------------------------

CREATE TEMP TABLE _valid_user_ids AS
SELECT
    u.user_id,
    u.signup_at
FROM "user" AS u
WHERE datetime(replace(trim(u.signup_at), ' KST', ''))
      >= datetime('2026-08-28 18:00:00');

CREATE UNIQUE INDEX temp.ux_valid_user_ids
    ON _valid_user_ids(user_id);

CREATE TABLE user_filtered AS
SELECT u.*
FROM "user" AS u
INNER JOIN _valid_user_ids AS vu
    ON vu.user_id = u.user_id;

-- ------------------------------------------------------------
-- 2. EVENT 정제
--    선정된 사용자의 이벤트 중 가입 이후 이벤트만 유지한다.
-- ------------------------------------------------------------

CREATE TABLE event_filtered AS
SELECT e.*
FROM event AS e
INNER JOIN _valid_user_ids AS vu
    ON vu.user_id = e.user_id
WHERE datetime(replace(trim(e.event_at), ' KST', ''))
      >= datetime(replace(trim(vu.signup_at), ' KST', ''));

-- ------------------------------------------------------------
-- 3. 세션별 유일 사용자 판정
--    기존 PowerShell과 동일하게 정제 전 전체 EVENT를 사용한다.
--    session_id 또는 user_id가 NULL/공백인 행은 제외한다.
--    정확히 한 명의 사용자에게만 연결된 세션을 남긴다.
-- ------------------------------------------------------------

CREATE TEMP TABLE _valid_session_owner AS
SELECT
    e.session_id,
    MIN(e.user_id) AS user_id
FROM event AS e
WHERE e.session_id IS NOT NULL
  AND trim(e.session_id) <> ''
  AND e.user_id IS NOT NULL
  AND trim(e.user_id) <> ''
GROUP BY e.session_id
HAVING COUNT(DISTINCT e.user_id) = 1;

CREATE UNIQUE INDEX temp.ux_valid_session_owner
    ON _valid_session_owner(session_id);

-- ------------------------------------------------------------
-- 4. SEARCH 정제
--    ① EVENT에서 세션 소유자를 확인할 수 있고
--    ② 세션 소유자가 정확히 한 명이며
--    ③ 그 사용자가 기준 USER 코호트에 속하고
--    ④ 검색 시각이 가입 시각 이상인 검색만 유지한다.
-- ------------------------------------------------------------

CREATE TABLE search_filtered AS
SELECT s.*
FROM search AS s
INNER JOIN _valid_session_owner AS so
    ON so.session_id = s.session_id
INNER JOIN _valid_user_ids AS vu
    ON vu.user_id = so.user_id
WHERE datetime(replace(trim(s.search_time), ' KST', ''))
      >= datetime(replace(trim(vu.signup_at), ' KST', ''));

CREATE TEMP TABLE _valid_search_ids AS
SELECT search_id
FROM search_filtered;

CREATE UNIQUE INDEX temp.ux_valid_search_ids
    ON _valid_search_ids(search_id);

-- ------------------------------------------------------------
-- 5. SEARCH_RESULT 및 SEARCH_FILTER 연쇄 정제
--    정제 후 살아남은 search_id에 연결된 행만 유지한다.
-- ------------------------------------------------------------

CREATE TABLE search_result_filtered AS
SELECT sr.*
FROM search_result AS sr
INNER JOIN _valid_search_ids AS vs
    ON vs.search_id = sr.search_id;

CREATE TABLE search_filter_filtered AS
SELECT sf.*
FROM search_filter AS sf
INNER JOIN _valid_search_ids AS vs
    ON vs.search_id = sf.search_id;

-- ------------------------------------------------------------
-- 6. BOOKING 정제
--    선정된 사용자의 가입 이후 확정 예약만 유지한다.
-- ------------------------------------------------------------

CREATE TABLE booking_filtered AS
SELECT b.*
FROM booking AS b
INNER JOIN _valid_user_ids AS vu
    ON vu.user_id = b.user_id
WHERE datetime(replace(trim(b.booking_at), ' KST', ''))
      >= datetime(replace(trim(vu.signup_at), ' KST', ''))
  AND b.booking_status = 'confirmed';

-- ------------------------------------------------------------
-- 7. 결과 테이블 기본키 중복 방지
--    CREATE TABLE AS SELECT는 원본 PK 제약을 복사하지 않으므로
--    UNIQUE INDEX로 결과 키의 유일성을 검증한다.
--    중복이 있으면 인덱스 생성 단계에서 실행이 실패한다.
-- ------------------------------------------------------------

CREATE UNIQUE INDEX ux_user_filtered_user_id
    ON user_filtered(user_id);

CREATE UNIQUE INDEX ux_event_filtered_event_id
    ON event_filtered(event_id);

CREATE UNIQUE INDEX ux_search_filtered_search_id
    ON search_filtered(search_id);

CREATE UNIQUE INDEX ux_search_result_filtered_id
    ON search_result_filtered(search_result_id);

CREATE UNIQUE INDEX ux_search_filter_filtered_id
    ON search_filter_filtered(search_filter_id);

CREATE UNIQUE INDEX ux_booking_filtered_booking_id
    ON booking_filtered(booking_id);

-- 관계 조회 성능용 인덱스
CREATE INDEX ix_event_filtered_user_id
    ON event_filtered(user_id);

CREATE INDEX ix_search_filtered_session_id
    ON search_filtered(session_id);

CREATE INDEX ix_search_result_filtered_search_id
    ON search_result_filtered(search_id);

CREATE INDEX ix_search_filter_filtered_search_id
    ON search_filter_filtered(search_id);

CREATE INDEX ix_booking_filtered_user_id
    ON booking_filtered(user_id);

COMMIT;
```

## 4. 결과 행 수 확인 SQL

최초 `2026-09-01 v01` 정제 결과와 같다면 다음 행 수가 나와야 한다.

```sql
SELECT 'user' AS table_name, COUNT(*) AS row_count
FROM user_filtered

UNION ALL

SELECT 'event', COUNT(*)
FROM event_filtered

UNION ALL

SELECT 'search', COUNT(*)
FROM search_filtered

UNION ALL

SELECT 'search_result', COUNT(*)
FROM search_result_filtered

UNION ALL

SELECT 'search_filter', COUNT(*)
FROM search_filter_filtered

UNION ALL

SELECT 'booking', COUNT(*)
FROM booking_filtered;
```

기대값:

| 테이블 | 기대 행 수 |
|---|---:|
| `user_filtered` | 89 |
| `event_filtered` | 10,432 |
| `search_filtered` | 296 |
| `search_result_filtered` | 8,555 |
| `search_filter_filtered` | 296 |
| `booking_filtered` | 36 |

`hotel` 1,000행과 `room` 3,000행은 정제 대상이 아니므로 원본 테이블을 그대로 사용한다.

## 5. 관계 및 시간 조건 검증 SQL

다음 검사의 결과는 모두 `0`이어야 한다.

```sql
-- 1. 기준 시각 이전 가입 사용자가 남아 있는지 확인
SELECT COUNT(*) AS users_before_boundary
FROM user_filtered
WHERE datetime(replace(trim(signup_at), ' KST', ''))
      < datetime('2026-08-28 18:00:00');

-- 2. USER에 없는 EVENT가 남아 있는지 확인
SELECT COUNT(*) AS events_missing_user
FROM event_filtered AS e
LEFT JOIN user_filtered AS u
    ON u.user_id = e.user_id
WHERE u.user_id IS NULL;

-- 3. 가입 전 EVENT가 남아 있는지 확인
SELECT COUNT(*) AS events_before_signup
FROM event_filtered AS e
INNER JOIN user_filtered AS u
    ON u.user_id = e.user_id
WHERE datetime(replace(trim(e.event_at), ' KST', ''))
      < datetime(replace(trim(u.signup_at), ' KST', ''));

-- 4. USER에 없는 BOOKING이 남아 있는지 확인
SELECT COUNT(*) AS bookings_missing_user
FROM booking_filtered AS b
LEFT JOIN user_filtered AS u
    ON u.user_id = b.user_id
WHERE u.user_id IS NULL;

-- 5. 가입 전 BOOKING이 남아 있는지 확인
SELECT COUNT(*) AS bookings_before_signup
FROM booking_filtered AS b
INNER JOIN user_filtered AS u
    ON u.user_id = b.user_id
WHERE datetime(replace(trim(b.booking_at), ' KST', ''))
      < datetime(replace(trim(u.signup_at), ' KST', ''));

-- 6. confirmed가 아닌 예약이 남아 있는지 확인
SELECT COUNT(*) AS bookings_not_confirmed
FROM booking_filtered
WHERE booking_status <> 'confirmed'
   OR booking_status IS NULL;

-- 7. 부모 SEARCH가 없는 SEARCH_RESULT 확인
SELECT COUNT(*) AS orphan_search_results
FROM search_result_filtered AS sr
LEFT JOIN search_filtered AS s
    ON s.search_id = sr.search_id
WHERE s.search_id IS NULL;

-- 8. 부모 SEARCH가 없는 SEARCH_FILTER 확인
SELECT COUNT(*) AS orphan_search_filters
FROM search_filter_filtered AS sf
LEFT JOIN search_filtered AS s
    ON s.search_id = sf.search_id
WHERE s.search_id IS NULL;
```

## 6. 기본키 중복 검사 SQL

UNIQUE INDEX가 정상적으로 생성됐다면 중복은 없어야 한다. 별도로 확인하려면 다음 쿼리를 사용한다.

```sql
SELECT 'user' AS table_name,
       COUNT(*) - COUNT(DISTINCT user_id) AS duplicate_keys
FROM user_filtered

UNION ALL

SELECT 'event',
       COUNT(*) - COUNT(DISTINCT event_id)
FROM event_filtered

UNION ALL

SELECT 'search',
       COUNT(*) - COUNT(DISTINCT search_id)
FROM search_filtered

UNION ALL

SELECT 'search_result',
       COUNT(*) - COUNT(DISTINCT search_result_id)
FROM search_result_filtered

UNION ALL

SELECT 'search_filter',
       COUNT(*) - COUNT(DISTINCT search_filter_id)
FROM search_filter_filtered

UNION ALL

SELECT 'booking',
       COUNT(*) - COUNT(DISTINCT booking_id)
FROM booking_filtered;
```

## 7. 한 번에 보는 검증 요약 SQL

```sql
WITH checks AS (
    SELECT
        'users_before_boundary' AS check_name,
        COUNT(*) AS issue_count
    FROM user_filtered
    WHERE datetime(replace(trim(signup_at), ' KST', ''))
          < datetime('2026-08-28 18:00:00')

    UNION ALL

    SELECT
        'events_missing_user',
        COUNT(*)
    FROM event_filtered AS e
    LEFT JOIN user_filtered AS u
        ON u.user_id = e.user_id
    WHERE u.user_id IS NULL

    UNION ALL

    SELECT
        'events_before_signup',
        COUNT(*)
    FROM event_filtered AS e
    INNER JOIN user_filtered AS u
        ON u.user_id = e.user_id
    WHERE datetime(replace(trim(e.event_at), ' KST', ''))
          < datetime(replace(trim(u.signup_at), ' KST', ''))

    UNION ALL

    SELECT
        'bookings_missing_user',
        COUNT(*)
    FROM booking_filtered AS b
    LEFT JOIN user_filtered AS u
        ON u.user_id = b.user_id
    WHERE u.user_id IS NULL

    UNION ALL

    SELECT
        'bookings_before_signup',
        COUNT(*)
    FROM booking_filtered AS b
    INNER JOIN user_filtered AS u
        ON u.user_id = b.user_id
    WHERE datetime(replace(trim(b.booking_at), ' KST', ''))
          < datetime(replace(trim(u.signup_at), ' KST', ''))

    UNION ALL

    SELECT
        'bookings_not_confirmed',
        COUNT(*)
    FROM booking_filtered
    WHERE booking_status <> 'confirmed'
       OR booking_status IS NULL

    UNION ALL

    SELECT
        'orphan_search_results',
        COUNT(*)
    FROM search_result_filtered AS sr
    LEFT JOIN search_filtered AS s
        ON s.search_id = sr.search_id
    WHERE s.search_id IS NULL

    UNION ALL

    SELECT
        'orphan_search_filters',
        COUNT(*)
    FROM search_filter_filtered AS sf
    LEFT JOIN search_filtered AS s
        ON s.search_id = sf.search_id
    WHERE s.search_id IS NULL
)
SELECT
    check_name,
    issue_count,
    CASE
        WHEN issue_count = 0 THEN 'PASS'
        ELSE 'FAIL'
    END AS status
FROM checks
ORDER BY check_name;
```

## 8. 실행 취소 및 오류 처리

`BEGIN IMMEDIATE`부터 `COMMIT`까지 실행하는 도중 오류가 발생하면 다음 명령으로 트랜잭션을 취소한다.

```sql
ROLLBACK;
```

UNIQUE INDEX 생성에서 실패한다면 원본에 기본키 중복이 있다는 뜻이다. 이 경우 임의로 중복을 삭제하지 말고 중복 키와 우선순위 규칙을 먼저 확인한다.

```sql
-- 예: USER 중복 키 확인
SELECT
    user_id,
    COUNT(*) AS row_count
FROM "user"
GROUP BY user_id
HAVING COUNT(*) > 1;
```

## 9. 현재 v03 데이터 사용 시 주의사항

현재 분석 기준본은 `2026-09-03 v03 비식별` 데이터다. 위 정제 로직과 별도로 지표 계산 시 다음 품질 플래그를 적용해야 한다.

### 숙박 날짜 기반 분석

```sql
SELECT *
FROM search_filtered
WHERE invalid_stay_date_flag = 0;
```

- 숙박일수, 체크아웃 날짜, 숙박기간, 숙박 날짜 코호트는 `invalid_stay_date_flag = 0`만 사용한다.
- 결과 없음, 재검색, 클릭 존재 등 검색 시각 기반 행동 분석에서는 플래그가 1인 행을 전체 데이터에서 삭제하지 않는다.

### 노출 기반 클릭 분석

```sql
SELECT *
FROM event_filtered
WHERE event_type = 'hotel_click'
  AND click_in_result_flag = 1;
```

- 노출 기반 클릭률과 검색결과 호텔 귀속 분석은 `click_in_result_flag = 1`인 `hotel_click`만 사용한다.
- 전체 검색의 클릭 존재 또는 이벤트 순서 분석에서는 플래그가 0인 클릭도 유지하되, 검색결과 노출 연결이 검증되지 않았음을 명시한다.

## 10. 실행 예시

DB Browser for SQLite, DBeaver 또는 SQLite를 지원하는 SQL 편집기에서 다음 순서로 실행한다.

1. 원본 SQLite 파일을 복사한다.
2. 복사한 데이터베이스를 연다.
3. 이 문서의 **전체 실행 SQL**을 실행한다.
4. **결과 행 수 확인 SQL**을 실행한다.
5. **한 번에 보는 검증 요약 SQL**을 실행한다.
6. 기대 행 수와 다르거나 `FAIL`이 발생하면 결과를 사용하지 않는다.
